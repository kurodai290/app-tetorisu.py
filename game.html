const canvas = document.getElementById('canvas'); const ctx = canvas.getContext('2d'); ctx.scale(20, 20); let score = 0; let level = 1; let gameRunning = false; let dropCounter = 0; let dropInterval = 1000; let lastTime = 0; let animationId = null; // アニメーションを管理するIDを追加 function updateInfo() { document.getElementById('score').innerText = score; document.getElementById('level').innerText = level; } // 壁判定のバグを修正（画面外の参照を防ぐ条件を追加） function collide(arena, player) { const [m, o] = [player.matrix, player.pos]; for (let y = 0; y < m.length; ++y) { for (let x = 0; x < m[y].length; ++x) { if (m[y][x] !== 0) { // 画面の下、左、右の壁を越えているかチェック if (!arena[y + o.y] || arena[y + o.y][x + o.x] === undefined || arena[y + o.y][x + o.x] !== 0) { return true; } } } } return false; } function createMatrix(w, h) { const matrix = []; while (h--) { matrix.push(new Array(w).fill(0)); } return matrix; } // 横幅12、縦20のマトリックス let arena = createMatrix(12, 20); const player = { pos: {x: 0, y: 0}, matrix: null }; function createPiece(type) { if (type === 'I') { return [[0,1,0,0],[0,1,0,0],[0,1,0,0],[0,1,0,0]]; } else if (type === 'L') { return [[0,2,0],[0,2,0],[0,2,2]]; } else if (type === 'J') { return [[0,3,0],[0,3,0],[3,3,0]]; } else if (type === 'O') { return [[4,4],[4,4]]; } else if (type === 'Z') { return [[5,5,0],[0,5,5],[0,0,0]]; } else if (type === 'S') { return [[0,6,6],[6,6,0],[0,0,0]]; } else if (type === 'T') { return [[0,7,0],[7,7,7],[0,0,0]]; } } function draw() { ctx.fillStyle = '#111'; ctx.fillRect(0, 0, canvas.width, canvas.height); drawGrid(); drawMatrix(arena, {x: 0, y: 0}); drawMatrix(player.matrix, player.pos); } function drawGrid() { ctx.lineWidth = 0.03; ctx.strokeStyle = '#2d2d2d'; for (let x = 0; x <= 12; x++) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, 20); ctx.stroke(); } for (let y = 0; y <= 20; y++) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(12, y); ctx.stroke(); } } function drawMatrix(matrix, offset) { const colors = [null, '#0DC2FF', '#FF8E0D', '#3877FF', '#FFE138', '#FF0D72', '#0DFF72', '#F538FF']; matrix.forEach((row, y) => { row.forEach((value, x) => { if (value !== 0) { ctx.fillStyle = colors[value]; ctx.fillRect(x + offset.x, y + offset.y, 1, 1); ctx.lineWidth = 0.05; ctx.strokeStyle = '#111'; ctx.strokeRect(x + offset.x, y + offset.y, 1, 1); } }); }); } function arenaSweep() { let rowCount = 1; outer: for (let y = arena.length - 1; y > 0; --y) { for (let x = 0; x < arena[y].length; ++x) { if (arena[y][x] === 0) { continue outer; } } const row = arena.splice(y, 1).fill(0); arena.unshift(row); ++y; score += rowCount * 10; rowCount *= 2; } let newLevel = Math.floor(score / 100) + 1; if (newLevel > 10) newLevel = 10; if (newLevel !== level) { level = newLevel; dropInterval = 1000 - (level - 1) * 100; } updateInfo(); } function merge(arena, player) { player.matrix.forEach((row, y) => { row.forEach((value, x) => { if (value !== 0) { arena[y + player.pos.y][x + player.pos.x] = value; } }); }); } function playerDrop() { if (!gameRunning) return; player.pos.y++; if (collide(arena, player)) { player.pos.y--; merge(arena, player); playerReset(); arenaSweep(); } dropCounter = 0; } function playerHardDrop() { if (!gameRunning) return; while (!collide(arena, player)) { player.pos.y++; } player.pos.y--; merge(arena, player); playerReset(); arenaSweep(); dropCounter = 0; } function playerMove(dir) { if (!gameRunning) return; player.pos.x += dir; if (collide(arena, player)) { player.pos.x -= dir; } } function gameOver() { gameRunning = false; document.getElementById('overlay').style.display = 'flex'; if (animationId) { cancelAnimationFrame(animationId); // 重複防止のためにタイマーを止める } } function playerReset() { const pieces = 'ILJOTSZ'; player.matrix = createPiece(pieces[pieces.length * Math.random() | 0]); player.pos.y = 0; // 盤面の横幅(12)をベースに中央に配置 player.pos.x = (12 / 2 | 0) - (player.matrix.length / 2 | 0); if (collide(arena, player)) { gameOver(); } } function playerRotate(dir) { if (!gameRunning) return; const pos = player.pos.x; let offset = 1; rotate(player.matrix, dir); while (collide(arena, player)) { player.pos.x += offset; offset = -(offset + (offset > 0 ? 1 : -1)); if (offset > player.matrix.length) { rotate(player.matrix, -dir); player.pos.x = pos; return; } } } function rotate(matrix, dir) { for (let y = 0; y < matrix.length; ++y) { for (let x = 0; x < y; ++x) { [matrix[x][y], matrix[y][x]] = [matrix[y][x], matrix[x][y]]; } } if (dir > 0) { matrix.forEach(row => row.reverse()); } else { matrix.reverse(); } } // ゲームループの多重起動を防ぐ処理を追加 function update(time = 0) { if (!gameRunning) return; const deltaTime = time - lastTime; lastTime = time; dropCounter += deltaTime; if (dropCounter > dropInterval) { playerDrop(); } draw(); animationId = requestAnimationFrame(update); } // 【新規追加】ボタンを押したときに安全に初期化して開始する関数 function startGame() { // 1. 画面のゲームオーバー表示を消す document.getElementById('overlay').style.display = 'none'; // 2. 盤面とスコアをきれいにリセットする arena = createMatrix(12, 20); score = 0; level = 1; dropInterval = 1000; dropCounter = 0; lastTime = performance.now(); updateInfo(); // 3. 古いゲームのタイマーが動いていたら一度完全に止める if (animationId) { cancelAnimationFrame(animationId); } // 4. フラグをONにしてブロックを作り、ゲームをスタート gameRunning = true; playerReset(); update(); } // キーボード操作の受付 document.addEventListener('keydown', event => { if (!gameRunning) return; if (event.keyCode === 37) { playerMove(-1); } else if (event.keyCode === 39) { playerMove(1); } else if (event.keyCode === 40) { playerDrop(); } else if (event.keyCode === 38) { playerRotate(1); } else if (event.keyCode === 32) { playerHardDrop(); event.preventDefault(); } }); // ページを開いたときに自動で最初のゲームを開始する startGame();
